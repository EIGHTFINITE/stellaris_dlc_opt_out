civic_hive_deep_sleep = {
	playable = {
		has_nomads_dlc = yes
	}
	ai_playable = {
		has_nomads_dlc = yes
	}
	description = "civic_deep_sleep_effects"
	potential = {
		authority = {
			value = auth_hive_mind
		}
	}
	possible = {
		is_nomadic = yes
	}
	modification = {
		add = {
			always = yes
		}
		remove = {
			custom_tooltip = {
				fail_text = civic_deep_sleep_not_moddable
				NOT = {
					any_owned_colony = {
						has_deposit = d_deep_sleep_cryo_pods
					}
				}
			}
		}
	}
	modifier = {
		leader_lifespan_mult = 0.1
		councilor_gestalt_regulatory_exp_gain = @gestalt_civic_node_xp_rate
	}
}
